## Usage Examples

### Accessing Exam Data
```typescript
import { DATA } from './data';

// Get JEE Physics chapters
const jeePhysics = DATA.find(e => e.id === 'jee')
  ?.groups.find(g => g.subject === 'Physics')
  ?.chapters;

// Count total chapters
const totalChapters = DATA.reduce((sum, exam) => 
  sum + exam.groups.reduce((s, g) => s + g.chapters.length, 0), 0
);
```

### Accessing Formulas
```typescript
import { FORMULAS } from './formulas';

// Get Physics formulas
const physicsFormulas = FORMULAS.find(f => f.subject === 'Physics');

// Get specific topic formulas
const rotationalFormulas = physicsFormulas
  ?.categories.find(c => c.topic === 'Rotational Motion')
```
